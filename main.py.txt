import os

# 1. API Configurations & SECRETS Injected by GitHub
API_KEY = os.getenv("ODDS_API_KEY")
BASE_URL = "https://api.the-odds-api.com/v4/sports"
REGIONS = "us"             # Focusing on retail US books (DraftKings, FanDuel, etc)
MARKETS = "h2h,spreads"    # Pulling Moneyline and Point Spreads
ODDS_FORMAT = "american"

# The 5 Core Leagues we want to track
LEAGUES = [
    "americanfootball_nfl",

]

def fetch_league_odds(league_key):
    """Fetches real-time market data for an entire league."""
    url = f"{BASE_URL}/{league_key}/odds/"
    params = {
        "apiKey": API_KEY,
        "regions": REGIONS,
        "markets": MARKETS,
        "oddsFormat": ODDS_FORMAT
    }
    try:
        response = requests.get(url, params=params)
        if response.status_code == 200:
            return response.json()
        print(f"Error fetching {league_key}: Status code {response.status_code}")
        return []
    except Exception as e:
        print(f"Exception triggered on {league_key}: {e}")
        return []

        print("CRITICAL CRASH: ODDS_API_KEY environment variable is missing.")
        return

    for league in LEAGUES:
        raw_league_data = fetch_league_odds(league)
        all_raw_data.extend(raw_league_data)
